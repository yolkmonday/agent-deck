# Auto-updater + i18n Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Agent Deck gets an EN/ID UI (default English, switchable in Settings) and a self-updater fed by GitHub Releases.

**Architecture:** A dependency-free typed dictionary (`src/i18n`) with a Zustand `useLang` store persisted in localStorage; per-area dictionary files so string extraction can run in parallel. Updater = `tauri-plugin-updater` + `tauri-plugin-process`, a pure state reducer in `src/store/updater.ts`, a banner in the app shell and a section in `SettingsPopover`. CI signs updater artifacts and uploads `latest.json`.

**Tech Stack:** Tauri v2, React 19, Vite 8, TypeScript 6, Zustand 5, Tailwind v4, `bun test`.

**Spec:** `docs/superpowers/specs/2026-09-24-updater-i18n-design.md`

## Global Constraints

- Bun only (`bun add`, `bun test`, `bun run build`). No npm/yarn.
- No i18n library. Only new deps: `tauri-plugin-updater`, `tauri-plugin-process` (Rust) and `@tauri-apps/plugin-updater`, `@tauri-apps/plugin-process` (JS).
- Default language is `"en"` always; localStorage key `agent-deck.lang`; any other stored value → `"en"`.
- Keys are flat dotted strings `area.name` (e.g. `settings.language`). Placeholders `{name}`.
- Not translated: brand/product names, model ids, CLI commands, keyboard shortcut labels, `tok/s`, `ms`.
- EN units: `18.4M`, `448.7K`, `<1 min`, `26 min`, `1 h 4 min`, `12 s`. ID units unchanged from today: `18,4 jt`, `448,7 rb`, `<1 mnt`, `26 mnt`, `1 j 4 mnt`, `12 dtk`.
- Updater endpoint: `https://github.com/yolkmonday/agent-deck/releases/latest/download/latest.json`. Auto-check 10 s after start, then every 6 h, never when `import.meta.env.DEV`.
- Existing visual style: Tailwind tokens `bg-surface`, `bg-surface-2`, `border-border`, `text-fg`, `text-fg-2`, `text-fg-3`, `accent-waiting`, class `ad-interactive`. Reuse them.
- Commit messages: English, lowercase, concise, no Co-Authored-By line.
- Parallel workers (Tasks 3–6) do NOT commit; the controller commits after review.

## Review Focus

1. Corrupted/unknown value in localStorage (`"fr"`, `""`, JSON) → app boots in English, no crash. (Task 1 test)
2. localStorage unavailable (throws, e.g. bun test env) → `useLang` still initialises to `"en"` and `setLang` still switches. (Task 1 test)
3. Auto-check fails offline or with 404 (no `latest.json` published yet) → no banner, no error UI, status back to `idle`. (Task 8 test)
4. User clicks Later, 6 h periodic check finds the same version → banner stays hidden. (Task 8 test)
5. Switching language while a notification or formatted value is produced from non-React code (`notify.ts`, `format.ts`) → uses the new language immediately. (Task 2 test)

`src/i18n/index.ts`:

```ts
import { useCallback } from "react";
import { create } from "zustand";
import { en, type MessageKey } from "@/i18n/en";
import { id } from "@/i18n/id";

export type Lang = "en" | "id";
export type Params = Record<string, string | number>;
export type { MessageKey };

const STORAGE_KEY = "agent-deck.lang";
const dictionaries: Record<Lang, Record<MessageKey, string>> = { en, id };

export const isLang = (v: unknown): v is Lang => v === "en" || v === "id";

export const readStoredLang = (): Lang => {
  try {
    const v = localStorage.getItem(STORAGE_KEY);
    return isLang(v) ? v : "en";
  } catch {
    return "en";
  }
};

const applyDocumentLang = (lang: Lang) => {
  if (typeof document !== "undefined") document.documentElement.lang = lang;
};

export const useLang = create<{ lang: Lang; setLang: (lang: Lang) => void }>((set) => ({
  lang: readStoredLang(),
  setLang: (lang) => {
    try {
      localStorage.setItem(STORAGE_KEY, lang);
    } catch {
      // Preference is a convenience; switching must still work without storage.
    }
    applyDocumentLang(lang);
    set({ lang });
  },
}));

export const interpolate = (template: string, params?: Params): string =>
  params ? template.replace(/\{(\w+)\}/g, (m, k: string) => (k in params ? String(params[k]) : m)) : template;

export const translate = (lang: Lang, key: MessageKey, params?: Params): string =>
  interpolate(dictionaries[lang][key] ?? en[key], params);

/** For non-React code: reads the language at call time. */
export const t = (key: MessageKey, params?: Params): string => translate(useLang.getState().lang, key, params);

export const useT = () => {
  const lang = useLang((s) => s.lang);
  return useCallback((key: MessageKey, params?: Params) => translate(lang, key, params), [lang]);
};

export const locale = (lang: Lang = useLang.getState().lang): "en-US" | "id-ID" =>
  lang === "id" ? "id-ID" : "en-US";

export const syncDocumentLang = () => applyDocumentLang(useLang.getState().lang);
```

```ts
// format.ts
import { locale, t } from "@/i18n";

const oneDecimal = (n: number) =>
  new Intl.NumberFormat(locale(), { minimumFractionDigits: 1, maximumFractionDigits: 1 }).format(n);

export const formatTokens = (n: number): string => {
  if (n >= 1_000_000) return t("format.million", { n: oneDecimal(n / 1_000_000) });
  if (n >= 1_000) return t("format.thousand", { n: oneDecimal(n / 1_000) });
  return String(n);
};

export const formatDuration = (ms: number): string => {
  const minutes = Math.floor(ms / 60_000);
  if (minutes < 1) return t("format.lessThanMinute");
  if (minutes < 60) return t("format.minutes", { m: minutes });
  return t("format.hoursMinutes", { h: Math.floor(minutes / 60), m: minutes % 60 });
};

export const formatShort = (ms: number): string =>
  ms < 60_000 ? t("format.seconds", { s: Math.max(0, Math.floor(ms / 1000)) }) : formatDuration(ms);
```

Note: `Intl` with `useGrouping` default would print `1,234.5` for huge values; keep `useGrouping: false` to match today's output (`n.toFixed(1)` had no grouping). In `perf.ts` build the `Intl.NumberFormat` per call with `locale()` (keep grouping default there, as today) and use `t("format.seconds", { s: ... })` for TTFT seconds. Replace `"id-ID"` in the three `toLocale*` calls with `locale()`.

import { getVersion } from "@tauri-apps/api/app";
import { relaunch } from "@tauri-apps/plugin-process";
import { check } from "@tauri-apps/plugin-updater";

export interface FoundUpdate {
  version: string;
  notes: string | null;
  install: (onProgress: (p: number) => void) => Promise<void>;
}

export const findUpdate = async (): Promise<FoundUpdate | null> => {
  const update = await check();
  if (!update) return null;
  return {
    version: update.version,
    notes: update.body ?? null,
    install: async (onProgress) => {
      let total = 0;
      let received = 0;
      await update.downloadAndInstall((e) => {
        if (e.event === "Started") total = e.data.contentLength ?? 0;
        if (e.event === "Progress") {
          received += e.data.chunkLength;
          if (total > 0) onProgress(received / total);
        }
        if (e.event === "Finished") onProgress(1);
      });
    },
  };
};

export const restartApp = () => relaunch();
export const currentVersion = () => getVersion();

```ts
import { create } from "zustand";
import { findUpdate, restartApp, type FoundUpdate } from "@/lib/updater";

export type UpdateStatus = "idle" | "checking" | "none" | "available" | "downloading" | "error";
export interface UpdaterState {
  status: UpdateStatus;
  version: string | null;
  progress: number;
  error: string | null;
  dismissed: string | null;
  manual: boolean;
}
export type UpdaterEvent =
  | { type: "check"; manual: boolean }
  | { type: "found"; version: string }
  | { type: "none" }
  | { type: "failed"; error: string }
  | { type: "progress"; progress: number }
  | { type: "dismiss" };

export const initialUpdaterState: UpdaterState = {
  status: "idle",
  version: null,
  progress: 0,
  error: null,
  dismissed: null,
  manual: false,
};

export const reduce = (s: UpdaterState, e: UpdaterEvent): UpdaterState => {
  switch (e.type) {
    case "check":
      if (s.status === "downloading") return s;
      return { ...s, status: "checking", error: null, manual: e.manual };
    case "found":
      return { ...s, status: "available", version: e.version, progress: 0 };
    case "none":
      return { ...s, status: "none" };
    case "failed":
      return s.manual ? { ...s, status: "error", error: e.error } : { ...s, status: "idle", error: null };
    case "progress":
      return { ...s, status: "downloading", progress: Math.min(1, Math.max(0, e.progress)) };
    case "dismiss":
      return { ...s, dismissed: s.version };
  }
};

export const bannerVisible = (s: UpdaterState): boolean =>
  s.status === "downloading" || (s.status === "available" && s.version !== s.dismissed);

const AUTO_DELAY_MS = 10_000;
const AUTO_INTERVAL_MS = 6 * 60 * 60 * 1000;

let pending: FoundUpdate | null = null;

export const useUpdater = create<
  UpdaterState & { check: (manual: boolean) => Promise<void>; install: () => Promise<void>; dismiss: () => void }
>((set, get) => {
  const dispatch = (e: UpdaterEvent) => set(reduce(get(), e));
  return {
    ...initialUpdaterState,
    check: async (manual) => {
      if (get().status === "checking" || get().status === "downloading") return;
      dispatch({ type: "check", manual });
      try {
        pending = await findUpdate();
        dispatch(pending ? { type: "found", version: pending.version } : { type: "none" });
      } catch (err) {
        const error = err instanceof Error ? err.message : String(err);
        if (!manual) console.warn("[updater] check failed", error);
        dispatch({ type: "failed", error });
      }
    },
    install: async () => {
      if (!pending) return;
      dispatch({ type: "progress", progress: 0 });
      try {
        await pending.install((p) => dispatch({ type: "progress", progress: p }));
        await restartApp();
      } catch (err) {
        set({ ...get(), status: "error", manual: true, error: err instanceof Error ? err.message : String(err) });
      }
    },
    dismiss: () => dispatch({ type: "dismiss" }),
  };
});

export const startAutoCheck = (): (() => void) => {
  if (import.meta.env.DEV) return () => undefined;
  const run = () => void useUpdater.getState().check(false);
  const first = setTimeout(run, AUTO_DELAY_MS);
  const every = setInterval(run, AUTO_INTERVAL_MS);
  return () => {
    clearTimeout(first);
    clearInterval(every);
  };
};
```

Note: an install error sets `status: "error"` with `manual: true`, so the Settings section shows it; the banner hides (not `available`). The user can check again.
